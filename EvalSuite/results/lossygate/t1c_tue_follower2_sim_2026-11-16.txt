run() {         # dir, then overrides: the follower with the sim's defaults (night fits; no slot #1 process); FPID = its pid
  local S=$D/$1 s=$1; shift

    CHAIN_QDIR="$S/q" CHAIN_FREEGB="cat $S/free" CHAIN_MEMBUSY="test -e $S/membusy" CHAIN_LIDSHUT="test -e $S/lid" \

  FPID=$!

#!/bin/bash
# a stand-in for Leap's run_c25_tue1.sh: removes its marker at arming; writes it on any end (TERM: a stop by a signal)
MK=$1; TRIG=$2
rm -f "$MK"
trap 'echo "c25 end $(date "+%m-%d %H:%M:%S"): rc 1 (stopped by a signal)" > "$MK"; exit 1' TERM
while [ ! -e "$TRIG" ]; do sleep 0.2; done
echo "c25 end $(date '+%m-%d %H:%M:%S'): rc 0 (ran)" > "$MK"; exit 0
