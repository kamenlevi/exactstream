tc() { date -r $(( ( $(date +%s) + $1 + 59 ) / 60 * 60 )) '+%Y-%m-%d %H:%M'; }   # at least $1 s from now, minute-ceiled

alive() { local p; for p in "$@"; do [ -n "$p" ] && kill -0 "$p" 2>/dev/null && echo "ALIVE($p)"; done; }

}

# a stand-in for runner 65 / 66 under the REAL chain2 in the follower3 sim: records its settings and pid; refuses an existing
# run dir without CARD_RESUME=1 (as runner 65 does); makes the run dir; "runs" STUB_B2 / STUB_V2 s (a resume: STUB_RESUME
