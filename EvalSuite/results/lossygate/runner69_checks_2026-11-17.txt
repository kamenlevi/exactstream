def heredoc(tag):
    m = re.search(r"<<'%s'[^\n]*\n(.*?)\n%s\n" % (tag, tag), src, re.S)
    assert m, tag
    body = m.group(1).split("\n")
    while body and re.match(r"\s*\|\|", body[0]):   # a bash continuation line after the heredoc's start line
        body.pop(0)
    return "\n".join(body)
ENG2, RT, TM = heredoc("ENG2"), heredoc("RT"), heredoc("TM")

ok = bad = 0
def check(lab, got, want, out=""):
    global ok, bad
    if got == want:
        ok += 1; print(f"PASS {lab}")
    else:
        bad += 1; print(f"FAIL {lab}: got {got}, wanted {want}\n{out[-600:]}")
def run(code, env, *args):
    p = subprocess.run([sys.executable, "-", *args], input=code, text=True, capture_output=True, env={**os.environ, **env})
    return p.returncode, p.stdout + p.stderr
def runs(path, rs):
    json.dump(rs, open(path, "w"))
