#!/bin/bash
# Extract runner 68's (and runner 67's) Python heredocs and the runterms function verbatim, for the stub tests.
# A heredoc whose command line continues ("... 2>&1 \" then "    || void ...") starts after that continuation line.
R=$HOME/EvalSuite/tools; O=$(dirname "$0")
ext() { awk -v s="$2" 'index($0, "<<'"'"'" s "'"'"'") {p=1; c=($0 ~ /\\$/); next} p && c {c=($0 ~ /\\$/); next} p && $0 == s {p=0} p' "$1"; }
ext $R/day_runner68.sh ENG  > $O/r68_eng.py
ext $R/day_runner68.sh ENG2 > $O/r68_eng2.py
ext $R/day_runner68.sh RT   > $O/r68_rt.py
ext $R/day_runner67.sh ENG  > $O/r67_eng.py
awk '/^runterms\(\) \{/{p=1} p{print} p && /^}$/{exit}' $R/day_runner68.sh > $O/r68_runterms.sh
wc -l $O/r68_eng.py $O/r68_eng2.py $O/r68_rt.py $O/r67_eng.py $O/r68_runterms.sh

def check(name, got, want, out=""):

def fixture(name, files, rows, terms):
    """files: list of lists of runs; rows: keys in decode.jsonl. Returns the cwd of a sandbox with results/<run>/."""
    cwd = tempfile.mkdtemp(prefix=f"rt_{name}_", dir=D); run_ = "card_g1x_t1code"
    os.makedirs(f"{cwd}/results/{run_}")
    with open(f"{cwd}/results/{run_}/decode.jsonl", "w") as f:
        for k in rows:
            f.write(json.dumps({"key": k}) + "\n")
    with open(f"{cwd}/results/{run_}/engage_runs.txt", "w") as f:
        for i, runs in enumerate(files):
            p = f"{cwd}/p{i}_runs.json"; json.dump(runs, open(p, "w")); f.write(p + "\n")
    return cwd, run_

def bash_case(cwd, calls):
    sh = f"""set -u; cd '{cwd}'; LOG=log.txt; DONE=done.txt; TAG=g1cx45g1b; SCOPE=code; RUNTERMS='{COM}'

void() {{ say "$1: VOID: runner stops"; echo "CARD $TAG: VOID ($1)" > "$DONE"; exit 5; }}
source '{D}/r68_runterms.sh'
{calls}
"""

#!/bin/bash
# DRY runs of runner 68 (CARD_DRY=1: no lock, no GPU; it stops before the first GPU step) on g1b's nine envs, tier 0 and
# tier 1 in both orders, with G1's pins and each branch's CARD_ENGAGE. Output: one line per run + the DRY log's last lines.
cd "$HOME/EvalSuite" || exit 1
BMS=$HOME/BigMoE/data/stack; EX=ref2tnax,ref2tnaxn.r3,ref2tnaxn.r2
TREES=data/r57/snap/g1b=fb0ee18288dae0ac5528006557dfaf8487d63c3b,data/r57/snap/frozen_g1b=2584e36a814ae956ac5c2d9c9090c4d787298502,bigmoe=d92cdb4b9e81d386e4170323d50f11b34147aa5f
ONE_C=lossy_check:e3+edrop+lmsl_rows; RES=lossy_check:res_opt@run,lossy_check:opt_post@run; COM=lossy_check:com_rows@run,lossy_check:com_opt@run
blob() { case $1 in lx) echo 2e5b7167af865cc7867c2f1fd8165a5d40fdb9c8;; g1) echo b736d293176e83b12ecc4c8d18ad665d7c38ed4f;; g1r) echo e42de6d974f25572644109a299cd690522450b3c;;
  g1c) echo bfd66cedd73b692b4a9cabb62b49f5ac5a345f2c;; g1cr) echo 78f7932b9c72e6ece3f4af6eedd6651dbbd1e3bb;; g1cd) echo efb0de2be0b1ede610880bc460c3e87efcec440f;;
  g1cdr) echo f4eac1d6bd3705417c7d144c056ede62d3a588ae;; g1l) echo 1c3e2358e53d42d28421d622475de05392beb619;; g1lr) echo caab79159e3359af84dcb0e5f26582a93e838253;; esac; }
eng() { case $1 in lx) echo "";; g1|g1l) echo "$ONE_C";; g1r|g1lr) echo "$ONE_C,$RES";; g1c|g1cd) echo "$ONE_C,$COM";; g1cr|g1cdr) echo "$ONE_C,$COM,$RES";; esac; }
for e in lx g1 g1r g1c g1cr g1cd g1cdr g1l g1lr; do
  for mode in t0 t1 t1rev; do
    tier=1; ord=fwd; [ $mode = t0 ] && tier=0; [ $mode = t1rev ] && ord=rev
    if [ $e = lx ]; then ref=$EX; later=0; ident=0; else ref=card_lx45g1b_${tier/1/t1}code; [ $tier = 0 ] && ref=card_lx45g1b_t0code; [ $ord = rev ] && ref=${ref}_rev; later=1; ident=$tier; fi
    tg=${e}45g1b; [ $ord = rev ] && tg=${tg}_rev
    env CARD_DRY=1 CARD_TAG=${e}45g1b CARD_FENV=$BMS/${e}45g1b_env.txt CARD_FBLOB=$(blob $e) CARD_TIER=$tier CARD_ORDER=$ord CARD_SCOPE=code \
      "CARD_TREES=$TREES" CARD_SCREEN_SHA=6039fa8dbea5 CARD_SO_SHA=ff69fa24a745 "CARD_CUTOFF=2026-11-17 23:30" CARD_MEM=39 \
      CARD_REF=$ref CARD_REF_LATER=$later CARD_IFEVAL_IDENT=$ident CARD_ENGAGE="$(eng $e)" tools/day_runner68.sh > /dev/null 2>&1; rc=$?
    echo "$e $mode (ref $ref): rc $rc :: $(grep -E 'DRY: tier|runner stops' results/day68_dry_${tg}.log | tail -1 | cut -c10-400)"
  done
done
