########## the stand-in runner (runner_stub.sh)
#!/bin/bash
# a stand-in for runner 68 under the REAL tools/g1_chain.sh (byte copy) in the G1 chain sim: records its settings and pid;
# refuses an existing tier-1 run dir without CARD_RESUME=1 (as runner 68 does); makes the run dir; "runs" STUB_S s (a
# resume: STUB_RESUME s); writes the DONE file. Per-TG outcomes: STUB_KILL0 (tier 0 KILL), STUB_FAIL1 / STUB_VOID1 (a tier-1
# FAIL / VOID), STUB_RC6 (exit 6 on its FIRST call: the runner's own cutoff; no DONE). TERM: exit 143, no DONE.
H=${SIMD:?}; ORD=${CARD_ORDER:-fwd}; TG=$CARD_TAG; T=${CARD_TIER:-1}; RD=results/card_${CARD_TAG}_t${T}code
[ "$ORD" = rev ] && { TG=${TG}_rev; RD=${RD}_rev; }
echo "$(date +%H:%M:%S) runner pid $$ tg=$TG tier=$T resume=${CARD_RESUME:-0} dry=${CARD_DRY:-0} cutoff=$CARD_CUTOFF mem=$CARD_MEM fblob=$CARD_FBLOB fenv=$(basename "$CARD_FENV") ref=$CARD_REF ident=${CARD_IFEVAL_IDENT:-0} fwd_run=${CARD_FWD_RUN:-} engage=${CARD_ENGAGE:-} later=${CARD_REF_LATER:-0}" >> "$H/runner_calls.txt"
[ "${CARD_DRY:-0}" = 1 ] && exit 0
if [ "$T" = 1 ] && [ "${CARD_RESUME:-0}" != 1 ] && [ -e "$RD" ]; then echo "$(date +%H:%M:%S)   $RD exists without CARD_RESUME=1: stops (rc 2)" >> "$H/runner_calls.txt"; exit 2; fi
mkdir -p "$RD"
trap 'echo "$(date +%H:%M:%S)   $TG pid $$ got TERM" >> "$H/runner_calls.txt"; kill $! 2>/dev/null; exit 143' TERM
if [ "${CARD_RESUME:-0}" = 1 ]; then S=${STUB_RESUME:-4}; else S=${STUB_S:-4}; fi
eval "S=\${STUB_S_$(echo "$TG" | tr -c 'A-Za-z0-9\n' '_'):-$S}"
sleep "$S" & wait $!
if [ "${STUB_RC6:-}" = "$TG" ] && [ ! -e "$H/rc6_done_$TG" ]; then touch "$H/rc6_done_$TG"; echo "$(date +%H:%M:%S)   $TG: exit 6 (the runner's cutoff)" >> "$H/runner_calls.txt"; exit 6; fi
if [ "$T" = 0 ]; then
  if [ "${STUB_KILL0:-}" = "$TG" ]; then v="KILL (stub)"; else v="GO (stub)"; fi
  echo "LOSSY tier0 (CODE-ONLY mode): $v" > "results/CARD_${TG}_DONE"
else
  if [ "${STUB_VOID1:-}" = "$TG" ]; then echo "CARD $TG: VOID (stub)" > "results/CARD_${TG}_DONE"
  elif [ "${STUB_FAIL1:-}" = "$TG" ]; then echo "LOSSY tier1 (CODE-ONLY mode): FAIL; stub $TG" > "results/CARD_${TG}_DONE"
  else echo "LOSSY tier1 (CODE-ONLY mode): PASS; stub $TG" > "results/CARD_${TG}_DONE"; fi
fi
echo "$(date +%H:%M:%S)   $TG done: $(head -c 60 "results/CARD_${TG}_DONE")" >> "$H/runner_calls.txt"
exit 0
