"""Fixtures for certify98_dumpcheck.py --incapture: BigMoE's heplus24 sample (dumps + files_sha256.json) dressed as an
in-capture run, a slim dir of links to the dev-bank files, and one broken copy per test."""
import hashlib, json, os, shutil, sys
import numpy as np
D = os.path.dirname(os.path.abspath(__file__)); os.chdir(D)
SM = os.path.expanduser("~/BigMoE/data/deepdraft/certify98_sample"); CE = os.path.expanduser("~/BigMoE/data/deepdraft/ceiling")
BANK = os.path.expanduser("~/datasets/deepdraft/bank")
def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 24), b""): h.update(b)
    return h.hexdigest()
def jw(p, o): json.dump(o, open(p, "w"), indent=1, sort_keys=True)
M0 = json.load(open(f"{SM}/dumps_sha256.json")); F0 = json.load(open(f"{SM}/files_sha256.json")); names = sorted(F0["files"])
def base(files_json=f"{SM}/files_sha256.json", wd_names=names):
    M = json.loads(json.dumps(M0)); R = M["_run"]
    R["in_capture"] = True; R["capture_script_sha256"] = "ab" * 32
    R["wd_sha256"] = {n: hashlib.sha256(("wd " + n).encode()).hexdigest() for n in wd_names}
    R["files_sha256_json"] = sha(files_json)
    return M
def slimdir(d, swap=None, drop=None):
    shutil.rmtree(d, ignore_errors=True); os.makedirs(d)
    for n in names:
        if n == drop: continue
        if swap and n in swap: shutil.copy(swap[n], f"{d}/{n}")
        else: os.symlink(f"{BANK}/{n}", f"{d}/{n}")
def npz_with(src, dst, **arrays):
    z = np.load(src); a = {k: z[k] for k in z.files}; a.update(arrays); np.savez(dst, **a)
def entry(p, like):
    e = dict(M0[like]); e["sha256"] = sha(p); e["bytes"] = os.path.getsize(p); return e
TK, TR = f"{CE}/topk_t30b_heplus24.npz", f"{CE}/tree_t30b_heplus24.npz"
z = np.load(TK); t = np.load(TR); fl = [str(x) for x in z["files"]]

jw("m_b.json", base()); slimdir("slim")
M = base(); del M["_run"]["in_capture"]; jw("m_noic.json", M)
M = base(); M["_run"]["in_capture"] = "true"; jw("m_icstr.json", M)
M = base(); del M["_run"]["capture_script_sha256"]; jw("m_nocs.json", M)
jw("m_wdless.json", base(wd_names=names[1:]))
M = base(); M["_run"]["wd_sha256"]["tf_code_HumanEval_9999.npz"] = "cd" * 32; jw("m_wdextra.json", M)
M = base(); M["_run"]["wd_sha256"][names[0]] = "AB" * 32; jw("m_wdhex.json", M)
slimdir("slim_drop", drop=names[3])
# one ids token changed where a TOPK cell reads it (row + 1 of file 0's first cell)
i0 = fl.index(names[0]); r1 = int(z["row"][z["fi"] == i0][0]) + 1
ids = np.load(f"{BANK}/{names[0]}")["ids"].copy(); ids[r1] = ids[r1] + 1
npz_with(f"{BANK}/{names[0]}", "alt_tok.npz", ids=ids); slimdir("slim_tok", swap={names[0]: "alt_tok.npz"})
Ft = json.loads(json.dumps(F0)); Ft["files"][names[0]] = sha("slim_tok/" + names[0]); jw("files_tok.json", Ft); jw("m_tok.json", base("files_tok.json"))
# one file's ids cut so its last TOPK cell reads past the end
ids = np.load(f"{BANK}/{names[1]}")["ids"]; i1 = fl.index(names[1]); mx = int(z["row"][z["fi"] == i1].max())
npz_with(f"{BANK}/{names[1]}", "alt_cut.npz", ids=ids[: mx + 1]); slimdir("slim_cut", swap={names[1]: "alt_cut.npz"})
Fc = json.loads(json.dumps(F0)); Fc["files"][names[1]] = sha("slim_cut/" + names[1]); jw("files_cut.json", Fc); jw("m_cut.json", base("files_cut.json"))
# tree variants, each with its own manifest entry
a_fi, a_an = t["a_fi"], t["a_an"]
npz_with(TR, "tree_drop.npz", a_fi=a_fi[:-1], a_an=a_an[:-1], a_walk_start=t["a_walk_start"][:-1],
         root_top_ids=t["root_top_ids"][:-1], root_top_p=t["root_top_p"][:-1])
an2 = a_an.copy(); an2[0] += 1; npz_with(TR, "tree_shift.npz", a_an=an2)
npz_with(TR, "tree_order.npz", files=t["files"][::-1].copy())
y = t["y"].copy(); j = int(np.flatnonzero(y >= 0)[0]); y[j] = y[j] + 1; npz_with(TR, "tree_y.npz", y=y)
y = t["y"].copy(); j = int(np.flatnonzero(y == -1)[0]); y[j] = 0; npz_with(TR, "tree_pastend.npz", y=y)
M = base()
for n in ("tree_drop.npz", "tree_shift.npz", "tree_order.npz", "tree_y.npz", "tree_pastend.npz"):
    M[n] = entry(n, "tree_t30b_heplus24.npz")
jw("m_trees.json", M)
# files_sha256.json without one file (and wd_sha256 without it): the dumps name a file it lacks
Fl = json.loads(json.dumps(F0)); del Fl["files"][names[5]]; jw("files_less.json", Fl)

# a path, not a plain name, in files_sha256.json
Fp = json.loads(json.dumps(F0)); Fp["files"]["../slim/" + names[6]] = Fp["files"].pop(names[6]); jw("files_path.json", Fp)
jw("m_path.json", base("files_path.json", wd_names=sorted(Fp["files"])))
print("fixtures written:", sorted(os.listdir(D)))
