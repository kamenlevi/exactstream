### bt_b12.py
"""B12 (NewIdeas' PLAN_B candidate for one Eval suite decode verdict, Breakthrough 2026-10-21): frugal routing at ~10 % on
BigMoE's v3 tier engine, in a pinned copy (~/Breakthrough/v3f = BigMoE 7b8c7b4 bigmoe/ + ext/ as bigmoe_v3f, plus a
flag-mode loader: bit 20 = read and wait, bit 21 = late-eligible, unflagged = used only if resident; tierglu.py and hc8.py
pinned from MoonShots). Levers, each an env switch so the suite can run B12 minus one without a rebuild:
  B12_CM=51 B12_CL=20   MSB-3 base slots / LSB-1 sidecar slots per layer (plan 4433333333: ranks 0-1 at 4 bits, the rest
                        MSB-3; RAM ~ 48 x (CM x 2.15 + CL x 0.77) MB). Default CM = round(FRAC x 512).
  frugal (always)       router top-20 by score; ranks 0-1 read and waited for; the first 10 resident in rank order are
                        used (missing ranks 2-9 -> resident ranks 10-19), weights renormalised over those 10 (w2f).
  B12_HC=4 | 8 | 0      hyper-connection mixers (+ PLE K/V) quantised to 4 / 8 bits (MoonShots' patches/hc8); default 8
                        (8-bit: KLD +0.0000 hard / +0.0001 easy, -0.62 GB, not slower; 4-bit: +0.024 hard. Quantised mixers turn
                        BigMoE's HC fusions off; B12_HC=0 = bf16 mixers with the compiled HC).
  B12_LATE1=1 | 0       (default 0: no speed gain on v3) wait for rank 0 only; a rank 1 with either record part cold is read after phase A and its output
                        (on layer L's MoE input, rank 1's renormalised weight, at 4 bits) is written into the HC streams
                        after layer L+1's attention, before its MLP read (tf_frugal_var.py l1t2); default 1.
  B12_PF=1 [B12_SPARE=4] prefetch arm 1 (NewIdeas I8): layer L+1's corrected d=1 top-2 per row posted on L's base and
                        sidecar rows (both record parts), prefetched after L's demand reads (non-JIT), spares from C.
  B12_PRE=1 | 0         TrainLab's pre-routed frugal (replaces late rank 1; default 0 until validated): in layers 8-39
                        (B12_PRE_BAND) the two waited (4-bit) experts are layer L's router + MoEPrefetch's correction
                        on layer L-1's MoE input, top-2; the rest = the true router's best outside them, resident only;
                        weights = the TRUE router's probabilities over the chosen 10, renormalised. Here the pre-picks
                        ride in layer L's own row (quality-faithful; the one-layer slack needs an early post).
  B12_LATE3=2,4         "B8" (NewIdeas): MSB-3 ranks 2-3, when cold, are read after phase A and added half a layer
                        late (base pool only, BIGMOE_DEFER = 2 pairs); with FRAC 0.05 = B8 (frugal 5 % on v3).
Agreed configs (NewIdeas 22:05, HC8 default from 01:00): B12 = load_b12(0.10) with the defaults (8-bit mixers, no late);
  B8 = load_b12(0.05) with B12_LATE3=2,4; B8-fast = load_b12(0.05) with B12_W=3 (wait for ranks 0-2, no phase B).
Decode 1-row calls take this path; prefill and multi-row calls stay v3's exact paths. Note: the compiled module is named
bigmoe_ext like BigMoE's: import nothing from ~/BigMoE/bigmoe in the same process.
API (Eval suite): load_b12(frac=0.10) -> (rt, st).   CLI: bt_b12.py FRAC speed (PROMPT code|prose, BT_N) | check | gsmN"""
import glob, hashlib, json, os, re, sys, time

V3F = os.path.expanduser("~/Breakthrough/v3f")
K, KC = 10, 20
STASH, PENDH = {}, {}
CFG = {}


def _env():
    os.environ.setdefault("MTPLX_NGRAM_PREWARM", "off")
    for k in ("MTPLX_COMPILED_VERIFY", "MTPLX_QWEN4EXP_COMPILE", "MTPLX_COMPILE_AR_FORWARD", "MTPLX_COMPILED_TARGET_PREFIX"):
        os.environ.setdefault(k, "0")
    os.environ["BIGMOE_TIER"] = "4433333333"
    os.environ.setdefault("BIGMOE_TIER_GLU", "3c")
    os.environ["BIGMOE_SPARE"] = str(CFG["spare"]) if CFG["pf"] else "0"   # frugal: no background reads, unless B12_PF
    os.environ.pop("BIGMOE_PF_JIT", None)            # the loader turns JIT on if the variable exists at all
    os.environ["BIGMOE_PF_NOPUBLISH"] = "1"
    os.environ["BIGMOE_DEFER"] = str(CFG["l3"][1] - CFG["l3"][0]) if CFG["l3"] else ("1" if CFG["late1"] else "0")   # pairs


def load_b12(frac=0.10, mtp=False):
    cm = int(os.environ.get("B12_CM", "0")) or round(frac * 512)
    CFG.update(frac=frac, hc=int(os.environ.get("B12_HC", "8")), late1=os.environ.get("B12_LATE1", "0") == "1",
               cm=cm, cl=int(os.environ.get("B12_CL", "0")) or round(0.4 * cm),
               pre=os.environ.get("B12_PRE", "0") == "1",
               band=tuple(int(v) for v in os.environ.get("B12_PRE_BAND", "8,40").split(",")),
               l3=tuple(int(v) for v in os.environ["B12_LATE3"].split(",")) if os.environ.get("B12_LATE3") else None,
               w=int(os.environ.get("B12_W", "2")),
               pf=os.environ.get("B12_PF") == "1", spare=int(os.environ.get("B12_SPARE", "4")))
    assert CFG["w"] >= 2 and (not CFG["l3"] or CFG["l3"][0] >= CFG["w"]), (CFG["w"], CFG["l3"])
    if CFG["pre"]:
        CFG["late1"] = False                         # pre-routing replaces late rank 1 (both decide the waited pair)
    if CFG["l3"]:                                    # B8: MSB-3 ranks A..B-1 late (base pool only); not with late rank 1
        assert CFG["l3"][0] >= 2 and CFG["l3"][1] > CFG["l3"][0] and not CFG["late1"], CFG["l3"]
    _env()
    os.environ["BIGMOE_TIER_CM"], os.environ["BIGMOE_TIER_CL"] = str(CFG["cm"]), str(CFG["cl"])
    if CFG["hc"]:     # hc8 swaps the mixers for quantised ones: MTPLX's eager HC path only (hc8.py: not with BigMoE's HC
        for k in ("BIGMOE_COMPILE_HC", "BIGMOE_HC_VREAD", "BIGMOE_HC_PACK"):   # fusions; COMPILE_HC's kernels expect
            os.environ[k] = "0"                                                  # bf16 mixers: Metal build error)
    if V3F not in sys.path:
        sys.path.insert(0, V3F)
    import mlx.core as mx
    from bigmoe_v3f.loader import load_streamed
    from bigmoe_v3f import tier as T
    md = os.path.expanduser("~/BigMoE/models/bare")
    cfg = json.load(open(os.path.join(md, "config.json"))); cfg = cfg.get("text_config", cfg)
    NL = cfg["num_hidden_layers"]
    rt, st = load_streamed(md, CFG["cm"], mtp=mtp)
    model = rt.model if hasattr(rt, "model") else rt
    TM = model.language_model.model if hasattr(model, "language_model") else model.model
    for i, l in enumerate(TM.layers):
        l._b12_L = i                                  # layer index for the late writes (layer_call)
    CFG["routers"] = [l.mlp.gate for l in TM.layers]   # B12_PRE: layer L+1's router, applied at layer L
    if CFG["hc"]:
        import hc8                                   # v3f/hc8.py (MoonShots 3055bf7)
        print("b12: hc", hc8.install(model, bits=CFG["hc"]), flush=True)
        mx.clear_cache()
    _install(mx, T, NL)
    if os.environ.get("B12_CACHE_MB"):                # cap MLX's buffer cache AFTER loading (MTPLX's memory plan sets
        mx.set_cache_limit(int(os.environ["B12_CACHE_MB"]) << 20)   # its own at load); freed buffers count as RAM
    print(f"b12: C_m={CFG['cm']} C_l={CFG['cl']} ({100 * CFG['cm'] / 512:.0f} %), HC {CFG['hc'] or 'bf16'}, late rank 1 "
          f"{'on' if CFG['late1'] else 'off'}, late MSB-3 ranks {CFG['l3'] or 'none'}, pools {st.resident_gb():.2f} GB, "
          f"active {mx.get_active_memory() / 1e9:.2f} GB",
          flush=True)
    return rt, st


def _install(mx, T, NL):
    from mtplx.models import qwen4_exp as Q
    from mtplx.models.qwen4_exp import _hyper_residual_write
    from mtplx.attention_context import current_attention_phase
    from mtplx.attention_math import attention_gate
    TG = T._tier_glu()
    orig = Q.SparseMoeBlock.__call__                 # v3's tier block (prefill, multi-row): installed by load_streamed
    R = mx.arange(KC)
    W20, L21 = 1 << 20, 1 << 21
    PREP, CORR, ROUT = {}, [], []
    R22 = mx.arange(KC + 2)
    FL22 = mx.where(R22 < 2, W20, 0).astype(mx.int32)[None]          # pre-routed rows: the two pre-picks waited
    if CFG["pre"] or CFG["pf"]:                      # layer L+1's routers + MoEPrefetch's d=1 correction (source-indexed)
        import numpy as np
        z = np.load(os.path.expanduser(os.environ.get("BIGMOE_PF_CORR", "~/MoEPrefetch/data/corr125_r128.npz")))
        CORR = [(mx.array(z["A"][i]), mx.array(z["B"][i]), mx.array(z["mu"][i]), mx.array(z["delta"][i]))
                for i in range(z["A"].shape[0])]
        ROUT = CFG["routers"]
    A3, B3 = CFG["l3"] or (0, 0)
    W = CFG["w"]                                                      # B12_W: ranks < W read and waited for (default 2)
    FL = mx.where(R == 0, W20, mx.where(R == 1, L21 if CFG["late1"] else W20,
                                        mx.where(R < W, W20, mx.where((R >= A3) & (R < B3), L21, 0)))).astype(mx.int32)[None]
    NF = max(2, B3, W)                                                # ranks 0..NF-1 always among the 10 used
    FL_MR = mx.where(R < NF, W20, 0).astype(mx.int32)[None]          # verify rows: every forced rank waited
    from bigmoe_v3f.gated import MAXK
    SMAX = MAXK // KC                                                 # rows per call on the frugal path
    FL_ALL = mx.where(R < max(2, B3, CFG["w"]), W20, 0).astype(mx.int32)[None]   # last layer: nothing late, all forced waited
    WK_M, WK_L = 255 | (K << 8), 255 | (2 << 8)                       # flag mode; LRU-touch ranks < 10 / < 2

    def late3_fn(x2, wl, sml, dep_m, n_m, t, gate, L):
        """B8: the late MSB-3 ranks A3..B3-1 of layer L (base pool only), after layer L+1's attention."""
        Mp = B3 - A3

        def f(after):
            o = mx.isnan(after.reshape(-1)[:1]).astype(mx.int32) * 0
            pr = gate.spin(mx.concatenate([dep_m[:1] + o, mx.array([n_m, 0] * Mp, mx.int32)]), L, 2 * Mp, phase=1)
            p, s = pr[0::2], pr[1::2]                                 # (position, slot) pairs; pads (n, 0)
            match = p[None, :] == mx.arange(A3, B3)[:, None]          # [late ranks, pairs]
            sl_late = mx.where(match, s[None, :], 0).sum(axis=-1)     # each late rank's slot (0 if it was resident)
            ms = mx.where(sml[0] >= 0, sml[0], sl_late)[None]         # resident at phase A: weight 0 here anyway
            return TG(x2, t["base"], t["sideP"], ms, None, plan="3" * Mp, scores=wl).reshape(1, 1, -1)
        return f

    def late_fn(x2, wl, sm1, sl1, dep_m, dep_l, n_m, n_l, t, gate, L):
        def f(after):                                                 # spins ordered after layer L+1's attention
            o = mx.isnan(after.reshape(-1)[:1]).astype(mx.int32) * 0
            pm = gate.spin(mx.concatenate([dep_m[:1] + o, mx.array([n_m, 0], mx.int32)]), L, 2, phase=1)
            pl = gate.spin(mx.concatenate([dep_l[:1] + o, mx.array([n_l, 0], mx.int32)]), t["side"], 2, phase=1)
            ms = mx.where(sm1 >= 0, sm1, pm[1:2].reshape(1, 1))       # each record part: phase A's slot if it was in,
            ls = mx.where(sl1 >= 0, sl1, pl[1:2].reshape(1, 1))       # else the late read's
            return TG(x2, t["base"], t["sideP"], ms, ls, plan="4", scores=wl).reshape(1, 1, -1)   # weight 0 unless late
        return f

    def pre_routed(x, x2, gates, ids, pre, L, t, gate):
        """B12_PRE: the pre-picked pair (read and waited, 4-bit), then the true router's best resident outside it."""
        row = mx.concatenate([pre, ids], axis=-1)                     # [1, 22]: pre-picks, then the true top-20
        dep_m, n_m = gate.post(L, row | FL22, wk=WK_M)
        dep_l, n_l = gate.post(t["side"], pre | FL22[:, :2], wk=WK_L)
        sm = gate.spin(dep_m, L, n_m).reshape(1, KC + 2)
        sl = gate.spin(dep_l, t["side"], n_l).reshape(1, 2)
        dup = (ids == pre[:, 0:1]) | (ids == pre[:, 1:2])             # a pre-pick also in the true top-20: used once
        pool = (sm[:, 2:] >= 0) & ~dup
        selr = pool & (mx.cumsum(pool.astype(mx.int32), axis=-1) <= K - 2)
        sel = mx.concatenate([mx.ones((1, 2), mx.bool_), selr], axis=-1)
        pos = mx.argsort(mx.where(sel, R22, R22 + 1000), axis=-1)[:, :K]
        used = mx.take_along_axis(sel, pos, axis=-1)
        s10 = mx.where(used, mx.take_along_axis(sm, pos, axis=-1), sm[:, :1])
        ids10 = mx.take_along_axis(row, pos, axis=-1)
        w10 = mx.where(used, mx.take_along_axis(gates, ids10, axis=-1), 0.0)   # the TRUE router's probabilities
        w10 = w10 / w10.sum(axis=-1, keepdims=True)
        y = TG(x2, t["base"], t["sideP"], s10, sl, plan="".join(T.PLAN), scores=w10)
        return y.reshape(*x.shape[:-1], y.shape[-1])

    def block(self, x):
        bm = getattr(self.switch_mlp, "_bm", None)
        S = x.shape[-2] if x.ndim == 3 else 0
        if bm is None or "tier" not in bm or x.ndim != 3 or x.shape[0] != 1 or not 1 <= S <= SMAX:
            return orig(self, x)                                      # prefill / long verify: v3's exact tier paths
        L, t, gate = bm["L"], bm["tier"], bm["gate"]
        if L == bm["first"]:
            gate.new_pass()
        ae = bm.get("async_every", 0)
        if ae and L % ae == ae - 1:
            mx.async_eval(x)
        if current_attention_phase() == "decode_verify":             # v3's shared-expert form per phase (numerics)
            shared = attention_gate(self.shared_expert(x), self.shared_expert_gate(x))
        else:
            shared = mx.sigmoid(self.shared_expert_gate(x)) * self.shared_expert(x)
        x2 = x.reshape(S, x.shape[-1])
        gates = mx.softmax(self.gate(x2), axis=-1, precise=True)
        inds = mx.argpartition(gates, kth=-KC, axis=-1)[..., -KC:]
        sc = mx.take_along_axis(gates, inds, axis=-1)
        o = mx.argsort(-sc, axis=-1)
        ids = mx.take_along_axis(inds, o, axis=-1).astype(mx.int32)   # [S, 20] rank order per row
        p20 = mx.take_along_axis(sc, o, axis=-1)
        if CFG["pre"] and S == 1:
            b0, b1 = CFG["band"]
            if b0 <= L + 1 < b1 and L < len(CORR):                    # pick layer L+1's waited pair from this input
                A, Bm, mu, dl = CORR[L]
                x32 = x2.astype(mx.float32)
                lg = ROUT[L + 1](x2).astype(mx.float32) + ((x32 - mu) @ A.astype(mx.float32)) @ Bm.astype(mx.float32) + dl
                PREP[L + 1] = mx.argpartition(-lg, kth=1, axis=-1)[:, :2].astype(mx.int32)   # [1, 2]
            pre = PREP.pop(L, None)
            if pre is not None:
                return pre_routed(x, x2, gates, ids, pre, L, t, gate) + shared
        last = L == NL - 1
        one = S == 1
        fl = (FL_ALL if last else FL) if one else FL_MR               # verify rows: late-eligible ranks are waited for
        rows = (ids | fl).T.reshape(-1)                               # rank-major: every row's rank 0, then rank 1, ...
        pk = {}
        if CFG["pf"] and L + 1 < NL and L < len(CORR):               # B12_PF arm 1: layer L+1's corrected d=1 top-2 per
            A, Bm, mu, dl = CORR[L]                                   # row, prefetched (both record parts) after L's
            x32 = x2.astype(mx.float32)                               # demand reads (non-JIT loader, spares from C)
            lg = ROUT[L + 1](x2).astype(mx.float32) + ((x32 - mu) @ A.astype(mx.float32)) @ Bm.astype(mx.float32) + dl
            pr = mx.argpartition(-lg, kth=1, axis=-1)[:, :2].reshape(-1).astype(mx.int32)
            pk = dict(preds=pr, counts=(2 * S,))
        dep_m, n_m = gate.post(L, rows, wk=255 | ((K * S) << 8), **pk)   # flag mode; LRU-touch every row's ranks < 10
        dep_l, n_l = gate.post(t["side"], (ids[:, :2] | fl[:, :2]).T.reshape(-1), wk=255 | ((2 * S) << 8), **pk)
        sm = gate.spin(dep_m, L, n_m).reshape(KC, S).T                # [S, 20] base (MSB-3) slots, -1 = absent / late
        sl = gate.spin(dep_l, t["side"], n_l).reshape(2, S).T         # [S, 2] LSB sidecar slots of ranks 0-1
        forced = (R < NF)[None]                                       # ranks 0-1 (+ waited / late ranks) always counted
        pool = (sm >= 0) & ~forced
        sel = (pool & (mx.cumsum(pool.astype(mx.int32), axis=-1) <= K - NF)) | forced   # + first resident of the rest
        pos = mx.argsort(mx.where(sel, R, R + 1000), axis=-1)[:, :K]   # the 10 used positions, rank order (0, 1, ...)
        used = mx.take_along_axis(sel, pos, axis=-1)                  # fewer than 10 resident: the rest are padding
        s10 = mx.take_along_axis(sm, pos, axis=-1)
        s10 = mx.where(used, s10, s10[:, :1])                         # padding reads rank 0's slots at weight 0
        w10 = mx.where(used, mx.take_along_axis(p20, pos, axis=-1), 0.0)
        w10 = w10 / w10.sum(axis=-1, keepdims=True)                   # renormalised over the 10 used (w2f)
        l2 = sl
        if CFG["late1"] and not last and one:
            late = (sm[:, 1:2] < 0) | (sl[:, 1:2] < 0)                # rank 1 not in at phase A (either record part)
            s10 = mx.concatenate([s10[:, :1], mx.where(late, s10[:, :1], s10[:, 1:2]), s10[:, 2:]], axis=-1)
            l2 = mx.concatenate([sl[:, :1], mx.where(late, sl[:, :1], sl[:, 1:2])], axis=-1)
            w_now = mx.concatenate([w10[:, :1], mx.where(late, 0.0, w10[:, 1:2]), w10[:, 2:]], axis=-1)
            STASH[L] = late_fn(x2, mx.where(late, w10[:, 1:2], 0.0), sm[:, 1:2], sl[:, 1:2], dep_m, dep_l, n_m, n_l,
                               t, gate, L)
        elif CFG["l3"] and not last and one:                          # B8: cold ranks A3..B3-1 land half a layer late
            late = sm[:, A3:B3] < 0                                    # [1, B3-A3] (positions = ranks for r < NF)
            s10 = mx.concatenate([s10[:, :A3], mx.where(late, s10[:, :1], s10[:, A3:B3]), s10[:, B3:]], axis=-1)
            w_now = mx.concatenate([w10[:, :A3], mx.where(late, 0.0, w10[:, A3:B3]), w10[:, B3:]], axis=-1)
            STASH[L] = late3_fn(x2, mx.where(late, w10[:, A3:B3], 0.0), sm[:, A3:B3], dep_m, n_m, t, gate, L)
        else:
            w_now = w10
        y = TG(x2, t["base"], t["sideP"], s10, l2, plan="".join(T.PLAN), scores=w_now)   # [S, D] combined in-kernel
        return y.reshape(*x.shape[:-1], y.shape[-1]) + shared

    def layer_call(self, hidden, *, input_ids, ssm_mask, cache):     # MTPLX DecoderLayer.__call__ + the late writes
        if "ple" in self:
            hidden = hidden + self.ple(hidden, input_ids, cache)
        mixed, hyper, inject = self.attn_hyper_connection(hidden)
        if self.is_linear:
            block_out = self.linear_attn(mixed, ssm_mask, cache)
        else:
            block_out = self.self_attn(mixed, cache)
        L = self._b12_L
        for f, inj in PENDH.pop(L, []):                               # layer L-1's late rank 1, before this MLP read
            hyper = _hyper_residual_write(hyper, f(block_out), inj)
        mixed, hyper, inject = self.mlp_hyper_connection(hyper, pending=(block_out, inject))
        block_out = self.mlp(mixed)
        hidden = _hyper_residual_write(hyper, block_out, inject)
        f = STASH.pop(L, None)
        if f is not None:
            PENDH.setdefault(L + 1, []).append((f, inject))
        return hidden

    Q.SparseMoeBlock.__call__ = block
    if CFG["late1"] or CFG["l3"]:
        Q.DecoderLayer.__call__ = layer_call


# ---- D1i (TrainLab, adopted 04:35 Thu): the decoupled prefill as an API for the Eval suite and other harnesses.
# Prefill ids (all prompt tokens but the last) into a fresh model.make_cache(); attention in s_attn-token query
# chunks, the MoE once per s_moe tokens. Not bitwise vs plain chunked prefill (gather kernel shapes per group).
def b12_prefill(model, ids, cache, s_attn=512, s_moe=4096, keep=False):
    """D1i (3), TrainLab: per layer, the attention part (PLE, HC read, attention, HC write-into-read) in query chunks of
    s_attn tokens, then the MoE ONCE over the whole s_moe chunk (its experts read once per layer per s_moe tokens),
    then the HC residual write per query chunk. Same ops on the same shapes as a chunk-s_attn prefill except the MoE
    block, whose per-token outputs do not depend on the batch (grouped / sorted gather per pair, per-token combine),
    so it should equal B12_PREFILL_CHUNK=s_attn bit for bit while reading experts like chunk s_moe."""
    import mlx.core as mx
    import mtplx.models.qwen4_exp as Q
    from contextlib import nullcontext
    assert not (CFG["late1"] or CFG["l3"]), "decoupled prefill: MTPLX's own DecoderLayer only (no late writes)"
    tm = model.language_model.model
    P = ids.shape[1]
    scope = Q._rope_table_scope() if Q.qwen4_opdiet_enabled("rope") else nullcontext()
    st = lambda c: [a for a in (c.state if hasattr(c, "state") else []) if isinstance(a, mx.array)]
    SPLIT = os.environ.get("D1I_SPLIT", "1") == "1"

    def moe_split(blk, xs_):
        """mlx_lm Qwen3NextSparseMoeBlock.__call__ (the prefill path of the tier layers: bt_b12's block -> v3's tier
        block -> MTPLX SparseMoeBlock -> super) with router, shared expert and combine per query chunk (the
        matmul shapes of a chunk-s_attn prefill) and the routed experts ONCE over all chunks (grouped, per-pair)."""
        k, parts = blk.top_k, []
        for x in xs_:
            gates = mx.softmax(blk.gate(x), axis=-1, precise=True)
            inds = mx.argpartition(gates, kth=-k, axis=-1)[..., -k:]
            scores = mx.take_along_axis(gates, inds, axis=-1)
            if blk.norm_topk_prob:
                scores = scores / scores.sum(axis=-1, keepdims=True)
            parts.append((inds, scores))
        y = blk.switch_mlp(mx.concatenate(xs_, axis=1) if len(xs_) > 1 else xs_[0],

def b12_prefill_state(model, prompt_ids, s_attn=512, s_moe=4096):

    import mlx.core as mx

    cache = model.make_cache()

    kept = b12_prefill(model, ids, cache, s_attn, s_moe, keep=True)

def late_fn(xL, wv, dep, seq, n, sw, L):
    """phase B of layer L as a function of a later array (`after`): the spin is ordered after it"""
    base = type(sw).__mro__[1]

    def f(after):
        o = mx.isnan(after.reshape(-1)[:1]).astype(mx.int32) * 0
        depB = mx.concatenate([dep[:1] + o, mx.array([n, 0] * M, dtype=mx.int32)])
        lb = ext.spin_wait(depB, L, seq, 2 * M, 1)
        pos, slb = lb[0::2], lb[1::2]
        xf = xL.reshape(-1, xL.shape[-1])
        xpad = mx.concatenate([xf, mx.zeros((1, xf.shape[-1]), dtype=xf.dtype)])
        yB = base.__call__(sw, xpad[pos // n][None], slb[:, None][None].astype(mx.uint32))[0, :, 0, :]   # [M, D]
        wpad = mx.concatenate([wv.reshape(-1), mx.zeros((1,), wv.dtype)])                               # pads: weight 0
        return (yB * wpad[pos][:, None].astype(yB.dtype)).sum(axis=0).reshape(1, 1, -1)
    return f


def block(self, x):
    sw = self.switch_mlp
    bm = getattr(sw, "_bm", None)
    if bm is None or bm["full"] or x.shape[1] != 1:
        return orig_call(self, x)
    L = bm["L"]
    if L == bm["first"]:
        bm["gate"].new_pass()
    ae = bm.get("async_every", 0)
    if ae and L % ae == ae - 1:
        mx.async_eval(x)
    ids, pk = topk_probs(self.gate(x)[0], KC)                          # [1, KC] by score
    last = L == NL - 1
    fl = WALL if last else FLAGS
    use = None
    if THETA > 0 and not FRUGAL:                                       # mass rule: ranks past theta are not used at all
        p10 = pk[:, :K]
        use = ((mx.cumsum(p10, axis=-1) - p10) / p10.sum(axis=-1, keepdims=True)) < THETA
        fl = mx.where(use | (R < KW), fl, 0)
    flat = (ids.astype(mx.int32) | fl).reshape(-1)
    slots = bm["gate"].route_and_gate(L, flat, nw=255, nk=K).reshape(1, KC).astype(mx.int32)
    dep, seq, n = bm["gate"]._last
    res = slots >= 0
    if FRUGAL:                                                         # rank 0, rank 1 (now or late), then the first
        r1 = (R == 1)[None]                                            # resident ranks 2-19 up to 10 in all
        pool = res & ~r1
        sel = (pool & (mx.cumsum(pool.astype(mx.int32), axis=-1) <= K - 1)) | r1
        w = pk * sel
        w = w / w.sum(axis=-1, keepdims=True)                          # renormalised over the 10 (as select_resident)
    else:
        w = pk / pk.sum(axis=-1, keepdims=True)                        # no renormalisation beyond the top-10 sum
        if use is not None:
            w = w * use
    now = res                                                          # late ids are -1 here: phase B adds them
    safe = mx.where(res, slots, slots[:, :1]).astype(mx.uint32)        # rank 0 is always waited -> resident
    base = type(sw).__mro__[1]
    y = (base.__call__(sw, x, safe[None]) * (w * now)[None][..., None].astype(x.dtype)).sum(axis=-2)
    if not last and (LATE1 or (not FRUGAL and KW < K)):
        STASH[L] = late_fn(x, w, dep, seq, n, sw, L)
    return y + attention_gate(self.shared_expert(x), self.shared_expert_gate(x))


def layer_call(self, hidden, *, input_ids, ssm_mask, cache):          # MTPLX DecoderLayer.__call__ + the late writes
    if "ple" in self:
        hidden = hidden + self.ple(hidden, input_ids, cache)
    mixed, hyper, inject = self.attn_hyper_connection(hidden)
    if self.is_linear:
        block_out = self.linear_attn(mixed, ssm_mask, cache)
    else:
        block_out = self.self_attn(mixed, cache)
    L = self._bt_L
    for f, inj in PENDH.pop(L, []):                                   # layer L-1's late experts, before this MLP read
        hyper = _hyper_residual_write(hyper, f(block_out), inj)
    mixed, hyper, inject = self.mlp_hyper_connection(hyper, pending=(block_out, inject))
    block_out = self.mlp(mixed)
    hidden = _hyper_residual_write(hyper, block_out, inject)
    f = STASH.pop(L, None)
    if f is not None:
        if SAME:
            hidden = _hyper_residual_write(hidden, f(block_out), inject)
        else:
            PENDH.setdefault(L + 1, []).append((f, inject))
    return hidden

def block(self, x):
    sw = self.switch_mlp
    bm = getattr(sw, "_bm", None)
    S = x.shape[1]
    kc = min(20, MAXK // max(1, S))
    if bm is None or bm["full"] or S > SMAX or kc < K:
        CALLS["exact"] += 1
        return orig_call(self, x)
    CALLS["frugal"] += 1; CALLS["rows"] += S
    if bm["L"] == bm["first"]:
        bm["gate"].new_pass()
    ae = bm.get("async_every", 0)
    if ae and bm["L"] % ae == ae - 1:   # as the streamed SwitchGLU: hand the graph so far to the GPU
        mx.async_eval(x)
    ids, pk = topk_probs(self.gate(x)[0], kc)                     # [S, kc] router top-kc, softmax probs (descending)
    flat = ids.T.reshape(-1) if S > 1 else ids.reshape(-1)        # rank-major: every row's rank 0 first, then rank 1 ...
    preds, counts = None, (0, 0)
    nx = bm.get("next") if PF else None
    if nx is not None:   # BT_PF: look-ahead predictions for layers L+1.. (stream.py's unranked path) -> spare-slot prefetch
        rows = x.reshape(-1, x.shape[-1])
        parts, cnt, left = [], [], 30 + 31 * PRED_LINES
        for router, ratio, B in nx:
            b = min(B, left // rows.shape[0])
            if b <= 0:
                break
            lg = router(rows * ratio if ratio is not None else rows)
            if _CORR is not None and router is nx[0][0] and bm["L"] < len(_CORR):   # MoEPrefetch's corrected d=1
                _c = _CORR[bm["L"]]                                               # (BIGMOE_PF_CORR, as stream.py)
                lg = lg.astype(mx.float32) + ((rows.astype(mx.float32) - _c[2]) @ _c[0]) @ _c[1] + _c[3]
            parts.append(mx.argpartition(-lg, kth=b - 1, axis=-1)[:, :b].reshape(-1)); cnt.append(b * rows.shape[0])
            left -= cnt[-1]
        if parts:
            preds = mx.concatenate(parts) if len(parts) > 1 else parts[0]
            counts = tuple(cnt + [0] * (3 - len(cnt))) + (0,)
    slots = bm["gate"].route_and_gate(bm["L"], flat, preds=preds, counts=counts, nw=NW * S, nk=K * S)
    slots = (slots.reshape(kc, S).T if S > 1 else slots.reshape(1, kc)).astype(mx.int32)
    sl, w = select_resident(slots, pk, topk=K)                    # first K resident per row, renormalised
    base = type(sw).__mro__[1]
    y = (base.__call__(sw, x, sl[None].astype(mx.uint32)) * w[None][..., None].astype(x.dtype)).sum(axis=-2)
    return y + attention_gate(self.shared_expert(x), self.shared_expert_gate(x))

def gen(ids, n, conf, stop_eos=False):

    s.update(waited_per_tok=wm / nd, n=len(out))

def block(self, x):
    sw = self.switch_mlp
    bm = getattr(sw, "_bm", None)
    S = x.shape[1]
    kc = min(20, MAXK // max(1, S))
    if bm is None or bm["full"] or S > 8 or kc < K:
        return orig_call(self, x)
    if bm["L"] == bm["first"]:
        bm["gate"].new_pass()
    ae = bm.get("async_every", 0)
    if ae and bm["L"] % ae == ae - 1:
        mx.async_eval(x)
    ids, pk = topk_probs(self.gate(x)[0], kc)                           # [S, kc]
    p10 = pk[:, :K]; tot = p10.sum(axis=-1, keepdims=True)
    before = (mx.cumsum(p10, axis=-1) - p10) / tot                       # mass of the ranks before r
    th = mx.array(THETA) if ADA is None else mx.where(PREV_ENT["v"] <= ADA[2], ADA[0], ADA[1])
    wait = before < th                                                   # rank 0 always waited
    if S > 1 and S * K > C - 4:   # IDEAS D1d: a pass may not wait for more distinct experts than the pool holds
        M = max(0, C - 4 - S)                                            # every row's rank 0, then M more by priority
        prio = mx.where(wait, p10 / tot * (0.8 ** mx.arange(S, dtype=mx.float32))[:, None], -1.0)
        prio = mx.concatenate([mx.full((S, 1), -1.0), prio[:, 1:]], axis=-1)   # rank 0 is handled separately
        kth = mx.sort(prio.reshape(-1))[-M] if M > 0 else mx.array(9.0)
        wait = mx.concatenate([mx.ones((S, 1), mx.bool_), (prio[:, 1:] >= kth) & (prio[:, 1:] >= 0)], axis=-1)
    wait = wait.astype(mx.int32)
    flags = mx.concatenate([wait << 20, mx.zeros((S, kc - K), mx.int32)], axis=-1) if kc > K else wait << 20
    idf = ids.astype(mx.int32) | flags
    flat = idf.T.reshape(-1) if S > 1 else idf.reshape(-1)               # rank-major
    preds, counts = None, (0, 0)
    nx = bm.get("next") if PF else None
    if nx is not None:   # BT_PF: look-ahead predictions for layers L+1.. (stream.py's unranked path) -> spare-slot prefetch
        rows = x.reshape(-1, x.shape[-1])
        parts, cnt, left = [], [], 30 + 31 * PRED_LINES
        for router, ratio, B in nx:
            b = min(B, left // rows.shape[0])
            if b <= 0:
                break
            lg = router(rows * ratio if ratio is not None else rows)
            parts.append(mx.argpartition(-lg, kth=b - 1, axis=-1)[:, :b].reshape(-1)); cnt.append(b * rows.shape[0])
            left -= cnt[-1]
        if parts:
            preds = mx.concatenate(parts) if len(parts) > 1 else parts[0]
            counts = tuple(cnt + [0] * (3 - len(cnt))) + (0,)
    slots = bm["gate"].route_and_gate(bm["L"], flat, preds=preds, counts=counts, nw=255, nk=K * S)
    slots = (slots.reshape(kc, S).T if S > 1 else slots.reshape(1, kc)).astype(mx.int32)
    res = slots >= 0
    sel = res & (mx.cumsum(res.astype(mx.int32), axis=-1) <= K)         # first K resident in rank order
    w = pk * sel / tot                                                   # no renormalisation
    safe = mx.where(sel, slots, slots[:, :1]).astype(mx.uint32)
    base = type(sw).__mro__[1]
    y = (base.__call__(sw, x, safe[None]) * w[None][..., None].astype(x.dtype)).sum(axis=-2)
    return y + attention_gate(self.shared_expert(x), self.shared_expert_gate(x))

the rank is dropped. Strict LRU capacity C per layer (experts used in the pass become MRU). r=10 = exact routing.

def run(tr, C, r, K):
    fetch = subs = drops = 0
    for l in range(NL):
        c = OrderedDict()
        for a in range(0, tr.shape[0], K):
            used_all = []
            for t in range(a, min(a + K, tr.shape[0])):
                row = [int(x) for x in tr[t, l]]

                used = []
                for rank, e in enumerate(row[:10]):

                        if e not in c and e not in used_all: fetch += 1
                        used.append(e)
                    else:

                        if x is None: drops += 1
                        else: subs += 1; used.append(x); alt.remove(x)

                if e in c: c.move_to_end(e)
                else: c[e] = 1
            while len(c) > C: c.popitem(last=False)

def auroc(score, label):
    order = np.argsort(score); r = np.empty(len(score)); r[order] = np.arange(1, len(score) + 1)
    n1 = label.sum(); n0 = len(label) - n1
    return (r[label].sum() - n1 * (n1 + 1) / 2) / max(n1 * n0, 1)

def margin(d, top, tlp):
    hit = top == d[:, None]
    m = np.full(len(d), 99.0)
    rows = hit.any(1)
    m[rows] = tlp[rows, 0] - tlp[rows][hit[rows]]
    return m

def nwait(wt, rule):
    if rule[0] == "r":
        return int(rule[1:])
    return int(np.argmax(np.cumsum(wt) >= float(rule[1:]) - 1e-6)) + 1


def run(C, rule):
    loads, missed, ntok = 0, [], 0
    for k in ids:
        tr, wt = ids[k], w[k]
        ntok += tr.shape[0]
        for l in range(NL):
            c = OrderedDict()
            for t in range(tr.shape[0]):
                row, nw = tr[t, l], nwait(wt[t, l], rule)
                alt = [int(e) for e in row[10:20] if int(e) in c]
                used, miss = [], 0.0
                for r in range(10):
                    e = int(row[r])
                    if r < nw or e in c:
                        if e not in c:
                            loads += 1
                        used.append(e)
                    else:
                        miss += wt[t, l, r]
                        a = next((x for x in alt if x not in used), None)
                        if a is not None:
                            used.append(a)
                            alt.remove(a)
                missed.append(miss)
                for e in used:
                    if e in c:
                        c.move_to_end(e)
                    else:
                        c[e] = 1
                while len(c) > C:
                    c.popitem(last=False)
    m = np.array(missed)
    return loads / ntok, m.mean(), np.percentile(m, 99)

### mincore.py
"""How much of a set of files sits in the OS page cache (mincore over a read-only mmap). usage: mincore.py GLOB
(Breakthrough, 2026-10-20: checks whether F_NOCACHE expert reads could have been served from cached pages.)"""
import ctypes, ctypes.util, glob, mmap, os, sys
import numpy as np
libc = ctypes.CDLL(ctypes.util.find_library("c"), use_errno=True)
libc.mincore.argtypes = [ctypes.c_void_p, ctypes.c_size_t, ctypes.c_char_p]
PG = os.sysconf("SC_PAGE_SIZE")
tot = res = 0
for f in sorted(glob.glob(os.path.expanduser(sys.argv[1]))):
    sz = os.path.getsize(f)
    if sz == 0:
        continue
    fd = os.open(f, os.O_RDONLY)
    m = mmap.mmap(fd, sz, prot=mmap.PROT_READ)
    a = np.frombuffer(m, dtype=np.uint8)
    n = (sz + PG - 1) // PG
    vec = ctypes.create_string_buffer(n)
    if libc.mincore(ctypes.c_void_p(a.ctypes.data), ctypes.c_size_t(sz), vec) == 0:
        r = int((np.frombuffer(vec.raw, dtype=np.uint8) & 1).sum())
        tot += n; res += r
    del a
    m.close(); os.close(fd)
print(f"{sys.argv[1]}: {res * PG / 1e9:.1f} GB of {tot * PG / 1e9:.1f} GB in the page cache ({100 * res / max(1, tot):.1f} %)")

"""D2 first experiment: noise-injection sweep on the Qwen3.6-35B-A3B proxy (Breakthrough, 2026-10-19).
How much weight error (NMSE) can each tensor class take before it costs more than today's Q4 (MLX affine 4-bit g64)?

def segments(tok):

def logprobs(model, seg):
    out = model(seg[:, :-1])
    lg = (out[0] if isinstance(out, tuple) else out)[0].astype(mx.float32)

def ref_pass():

def modify(model, cls, arm, seed):
    if arm == "none":                      # null arm: reload only, nothing modified (measures the floor vs the ref pass)
        return 0, 0.0
    match = CLASSES[cls]
    mx.random.seed(seed)
    se = sw = 0.0
    n = 0
    for name, m in model.named_modules():
        if "language_model" not in name or not hasattr(m, "scales") or not hasattr(m, "bits") or not match(name):
            continue
        gs, bits, dt = m.group_size, m.bits, m.scales.dtype
        w = mx.dequantize(m.weight, m.scales, m.biases, group_size=gs, bits=bits).astype(mx.float32)
        if arm == "q4":
            ngs, nb, wn = 64, 4, w
        elif arm.startswith("t"):            # QTIP-style trellis fake-quant at k bits (scripts/trellis.py, L=12, CPU Viterbi)
            from trellis import quantize as tq
            ngs, nb = gs, bits
            wn = tq(w, int(arm[1:]), L=int(os.environ.get("BT_TL", "12")))
        else:
            ngs, nb = gs, bits
            wn = w + mx.random.normal(w.shape) * mx.sqrt(float(arm[1:]) * mx.mean(w * w, axis=-1, keepdims=True))
        wq, s, b = mx.quantize(wn.astype(dt), group_size=ngs, bits=nb)
        wd = mx.dequantize(wq, s, b, group_size=ngs, bits=nb).astype(mx.float32)
        e, t = mx.sum((wd - w) ** 2), mx.sum(w * w)
        mx.eval(wq, s, b, e, t)
        se += e.item()
        sw += t.item()
        n += 1
        m.weight, m.scales, m.biases, m.group_size, m.bits = wq, s, b, ngs, nb
        del w, wn, wd
        mx.clear_cache()
    return n, se / max(sw, 1e-30)


def compare(model, segs, ref):
    kl, agree, dnll, seg_kl = [], [], [], []
    for i, s in enumerate(segs):
        lp = logprobs(model, s)
        top, p_lp = mx.array(ref[f"top{i}"]), mx.array(ref[f"tlp{i}"])
        q_lp = mx.take_along_axis(lp, top, axis=-1)
        p, q = mx.exp(p_lp), mx.exp(q_lp)
        pt, qt = mx.maximum(1 - p.sum(-1), 1e-12), mx.maximum(1 - q.sum(-1), 1e-12)
        k = (p * (p_lp - q_lp)).sum(-1) + pt * (mx.log(pt) - mx.log(qt))
        nll = -mx.take_along_axis(lp, s[0, 1:, None], axis=-1)[:, 0]
        a1 = mx.argmax(lp, axis=-1)
        mx.eval(k, nll, a1)
        kl.append(np.array(k))
        seg_kl.append(float(kl[-1].mean()))
        agree.append(np.array(a1) == ref[f"a1{i}"])
        dnll.append(np.array(nll) - ref[f"nll{i}"])
    kl, agree, dnll = (np.concatenate(x) for x in (kl, agree, dnll))
    return kl.mean(), np.percentile(kl, 99), agree.mean(), dnll.mean(), seg_kl


def main():

        return ref_pass()
    cls, arms = sys.argv[1], sys.argv[2:]

                f"{seg_kl[0]:.4f}/{seg_kl[1]:.4f}/{seg_kl[2]:.4f})  top-1 agree {agree:.4f}  dNLL {dnll:+.4f}  "
                f"{time.time() - t0:.0f}s peak {mx.get_peak_memory() / 1e9:.1f} GB")
        print(line, flush=True)
        open(os.path.join(OUT, "results.txt"), "a").write(line + "\n")
        del model, segs

### pass_budget_sim.py
"""Pass-level expert budget (Breakthrough, 2026-10-19): a pass of K consecutive tokens may use at most U experts per
layer, chosen by rank-weighted votes of the rows' router top-10 (weight 1/(rank+1)). Reports the share of each row's
rank-0, rank-0/1 and top-10 choices inside the budget, and RAM bytes per token for that pass (U experts x 48 layers x
expert MB, plus dense) if all K tokens were accepted. 125B traces (MoEPrefetch traces125). Quality is NOT modelled.
"""
import numpy as np
T = "/Users/kamen/MoEPrefetch/data/traces125/"; SETS = ["code", "chat", "bulgarian", "math"]; NL = 48
trs = [np.stack([np.load(T + f"{s}_800.npz")[f"L{l}"] for l in range(NL)], 1).astype(np.int32)[:, :, :10] for s in SETS]
w = 1.0 / (np.arange(10) + 1)
def run(K, U):
    r0 = r01 = r10 = n = 0; uni = []
    for tr in trs:
        for a in range(0, tr.shape[0] - K + 1, K):
            blk = tr[a:a + K]
            for l in range(NL):
                ids = blk[:, l]; uni.append(len(np.unique(ids)))
                votes = {}
                for row in ids:
                    for r, e in enumerate(row): votes[e] = votes.get(e, 0) + w[r]
                keep = set(sorted(votes, key=votes.get, reverse=True)[:U])
                for row in ids:

    for l in range(NL):

                        continue
                    subs += 1

def replay(C, sets, delta=0.0, nw_rule=None):
    """delta >= 0: re-routing policy (all chosen read). nw_rule: mass-rule policy instead (wait ranks < nw, others used
    only if resident, no renorm). Returns reads/token, swapped-or-missed mass and energy (per layer, mean)."""
    reads = mass = en = 0.0; ntok = 0
    for k in sets:
        tr, g = ids[k], gp[k]
        ntok += tr.shape[0]
        for l in range(NL):
            c = OrderedDict()
            for t in range(tr.shape[0]):
                row, p = tr[t, l], g[t, l]
                w = p / p[:10].sum()
                res = np.array([int(e) in c for e in row])
                if nw_rule is not None:
                    nw = nw_rule(w[:10])
                    use = [r for r in range(10) if r < nw or res[r]]
                    miss = [r for r in range(10) if r >= nw and not res[r]]
                    mass += w[miss].sum(); en += (w[miss] ** 2).sum()
                else:
                    z = np.log(np.maximum(p, 1e-30)) + delta * res
                    z[:2] = np.inf                                  # true ranks 0-1 pinned
                    use = list(np.argsort(-z, kind="stable")[:10])
                    disp = [r for r in range(10) if r not in use]

                    if e in c:
                        c.move_to_end(e)
                    else:
                        c[e] = 1; reads += 1
                while len(c) > C:
                    c.popitem(last=False)
    n = ntok

def _make_mask(n):
    """MSB-n of every 4-bit code in a packed uint32 word: (codes with the low 4-n bits cleared, per-word sum of them)."""
    low = (1 << (4 - n)) - 1
    keep = mx.array(int(("%X" % (0xF & ~low)) * 8, 16), dtype=mx.uint32)

    @mx.compile
    def f(w):
        lsum = None
        for j in range(8):
            part = (w >> (4 * j)) & low
            lsum = part if lsum is None else lsum + part
        return w & keep, lsum
    return f


_MASK = {n: _make_mask(n) for n in (1, 2, 3)}


class P:   # bias-corrected MSB-n tier of a 4-bit slot pool (planes125's math, here for n = 1..3)
    @staticmethod
    def _tier_proj(lin, h, sl, idx, n):
        w, s, b = lin.weight[sl], lin.scales[sl], lin.biases[sl]
        gs = lin.group_size
        wm, lsum = _MASK[n](w)
        off = lsum.reshape(*s.shape, gs // 8).sum(-1).astype(mx.float32) / gs
        bias = (b.astype(mx.float32) + s.astype(mx.float32) * off).astype(s.dtype)
        return mx.gather_qmm(h, wm, s, bias, rhs_indices=idx, transpose=True, group_size=gs, bits=4)

def sync_slots(L, idl, nw, nk):
    """frugal residency without the GPU gate: (slots before, slots after) for the ids idl (-1 = absent). Loads the first nw
    ids (blocking), LRU-touches the resident ones among the first nk, never reads the others (BIGMOE_SPARE=0 semantics)."""
    pre = np.array(ext.lookup(L, idl))
    STAT["ld"] = STAT.get("ld", 0) + int((pre[:nw] < 0).sum()); STAT["c1"] = STAT.get("c1", 0) + 1   # measured reads
    need = idl[:nw] + [e for r, e in enumerate(idl[:nk]) if r >= nw and pre[r] >= 0]
    ext.ensure(L, mx.array(np.array(need, np.int32)))
    return pre, np.array(ext.lookup(L, idl))

def block(self, x):
    sw = self.switch_mlp
    bm = getattr(sw, "_bm", None)
    if bm is None or x.shape[1] != 1 or bm["full"]:
        return orig_call(self, x)
    if bm["L"] == bm["first"]:
        bm["gate"].new_pass()
    if PRE is not None and not MODE["exact"]:
        L = bm["L"]
        if PRE[0] <= L + 1 < PRE[1] and L < len(CORR):              # pre-pick layer L+1's two waited experts now
            x1 = x.reshape(1, -1)
            A, Bm, mu, dl = CORR[L]
            lg = ROUT[L + 1](x1).astype(mx.float32) + ((x1.astype(mx.float32) - mu) @ A) @ Bm + dl
            PREPICK[L + 1] = [int(v) for v in np.array(mx.argpartition(-lg, kth=1, axis=-1)[0, :2])]
        if L in PREPICK:
            return pre_block(self, sw, bm, x, PREPICK.pop(L))
    ids, pk = topk_probs(self.gate(x)[0], KC)                       # [1, KC]
    nw = NW
    if MODE["exact"] or (BAND is not None and not (BAND[0] <= bm["L"] < BAND[1])):   # exact: all top-10 waited
        _, slots = route(bm, ids, K, K)
        res = slots >= 0
        sel = res & (mx.cumsum(res.astype(mx.int32), axis=-1) <= K)
        w = pk * sel; w = w / mx.maximum(w.sum(axis=-1, keepdims=True), 1e-9)
        safe = mx.where(res, slots, slots[:, :1]).astype(mx.uint32)
        y = (type(sw).__mro__[1].__call__(sw, x, safe[None]) * w[None][..., None].astype(x.dtype)).sum(axis=-2)
        return y + attention_gate(self.shared_expert(x), self.shared_expert_gate(x))
    if THETA is not None:                                          # CPU sync per call: a quality probe, not a speed path
        th = THETA if ADA is None else (ADA["lo"] if LAST["ent"] <= ADA["tau"] else ADA["hi"])
        p = np.array(pk[0, :K].astype(mx.float32)); c = np.cumsum(p) / p.sum()
        nw = max(1, int(np.searchsorted(c, th - 1e-9)) + 1)
        STAT["w"] += nw; STAT["wn"] += 1
    if LATE is not None:                                           # late ranks are read too (only their timing differs)
        nw = max(nw, LATE[1])
    pre0 = None
    if TAILBITS and not SYNCGATE:                                  # residency before the call (CPU, no loading)
        pre0 = np.array(ext.lookup(bm["L"], [int(v) for v in np.array(ids[0, :K])]))
    pre_s, slots = route(bm, ids, nw, K)
    pre = pre0 if pre0 is not None else (pre_s[:K] if pre_s is not None else None)
    if SERE:
        return sere_block(self, sw, bm, x, ids, pk, slots)
    res = slots >= 0
    sel = res & (mx.cumsum(res.astype(mx.int32), axis=-1) <= K)      # first K resident in rank order
    w = pk * sel
    p10 = pk[:, :K].sum(axis=-1, keepdims=True) if self.norm_topk_prob else 1.0
    if BETA is not None:
        w = w / ((1 - BETA) * p10 + BETA * w.sum(axis=-1, keepdims=True))
    elif RENORM:
        w = w / mx.maximum(w.sum(axis=-1, keepdims=True), 1e-9)
    else:
        w = w / p10
    safe = mx.where(res, slots, slots[:, :1]).astype(mx.uint32)      # rank 0 is always resident (waited)
    base = type(sw).__mro__[1]
    yk = base.__call__(sw, x, safe[None])                             # [1, 1, KC, D]
    if TAILBITS:
        loaded = np.zeros(KC, bool); loaded[:K] = pre < 0; loaded[nw:] = False   # read this call: missing waited ranks
        low = loaded.copy(); low[:2] = False                          # ranks 0-1 stay 4-bit
        STAT["calls"] += 1; STAT["full"] += int(loaded[:2].sum()); STAT["low"] += int(low.sum())
        if low.any():
            xe = mx.expand_dims(x, (-2, -3)); sl = safe.reshape(-1)
            idx = mx.arange(KC).astype(mx.uint32).reshape(1, 1, KC)
            h = sw.activation(P._tier_proj(sw.up_proj, xe, sl, idx, TAILBITS), P._tier_proj(sw.gate_proj, xe, sl, idx, TAILBITS))
            ylow = P._tier_proj(sw.down_proj, h, sl, idx, TAILBITS).squeeze(-2)
            yk = mx.where(mx.array(low)[None, None, :, None], ylow.astype(yk.dtype), yk)
    if LATE is not None and bm["L"] < NLAY - 1:                    # cold ranks A..B-1: out of this layer's sum, into the
        r = np.arange(KC)                                          # HC streams after layer L+1's attention (layer_call)
        lm = (r >= LATE[0]) & (r < LATE[1]) & (np.concatenate([pre, np.zeros(KC - len(pre), int)]) < 0)
        if lm.any():
            lmx = mx.array(lm)[None]
            STASH[bm["L"]] = (yk * (w * lmx)[None][..., None].astype(x.dtype)).sum(axis=-2)
            w = w * ~lmx
            STAT["late"] = STAT.get("late", 0) + int(lm.sum())
    STAT["blk"] = STAT.get("blk", 0) + 1
    y = (yk * w[None][..., None].astype(x.dtype)).sum(axis=-2)
    return y + attention_gate(self.shared_expert(x), self.shared_expert_gate(x))


def pre_block(self, sw, bm, x, pre):
    """pre2: the pre-picked two are read and waited for (in an engine: issued a layer earlier); the rest = the true router's
    best outside them, resident only, up to 10 in all; weights = the true probabilities over the chosen, renormalised."""
    L = bm["L"]
    p = np.array(mx.softmax(self.gate(x)[0].astype(mx.float32), axis=-1, precise=True)[0])   # true router, all experts
    order = np.argsort(-p)
    rest = [int(e) for e in order[:KC + 2] if int(e) not in pre][:KC - 2]
    idl = list(pre) + rest
    _, after = sync_slots(L, idl, 2, K)                            # reads the pre-picks; LRU-touches resident ranks < 10
    use = [0, 1] + [i for i in range(2, len(idl)) if after[i] >= 0][:K - 2]
    ids_u = [idl[i] for i in use]
    w = p[ids_u].astype(np.float64); w = w / w.sum()
    top2, top10 = set(int(e) for e in order[:2]), set(int(e) for e in order[:K])
    STAT["pre_n"] = STAT.get("pre_n", 0) + 2
    STAT["pre_top2"] = STAT.get("pre_top2", 0) + sum(e in top2 for e in pre)
    STAT["pre_top10"] = STAT.get("pre_top10", 0) + sum(e in top10 for e in pre)
    b = L // 8                                                     # per 8-layer band: true rank 0 / rank 1 among the picks
    PB = STAT.setdefault("pre_band", {})
    c = PB.setdefault(b, [0, 0, 0])
    c[0] += 1; c[1] += int(order[0]) in pre; c[2] += int(order[1]) in pre
    sl = [int(after[i]) for i in use] + [int(after[0])] * (K - len(use))
    w = np.concatenate([w, np.zeros(K - len(use))]).astype(np.float32)
    base = type(sw).__mro__[1]
    y = (base.__call__(sw, x, mx.array(sl, mx.uint32).reshape(1, 1, K)) * mx.array(w)[None, None, :, None].astype(x.dtype)).sum(axis=-2)
    return y + attention_gate(self.shared_expert(x), self.shared_expert_gate(x))


def sere_block(self, sw, bm, x, ids, pk, slots):
    L = bm["L"]
    if L not in SIM:
        d = os.path.expanduser("~/Breakthrough/data/simtab")
        SIM[L] = (np.load(f"{d}/D{L}.npy").astype(np.float32), np.load(f"{d}/N{L}.npy"))
    Dm, Nm = SIM[L]
    i, sl, p = np.array(ids[0]), np.array(slots[0]), np.array(pk[0].astype(mx.float32))
    w10 = p[:K] / p[:K].sum()
    res = [(int(e), int(s)) for e, s in zip(i, sl) if s >= 0]
    used, outs, ws = set(), [], []
    for r in range(K):
        if sl[r] >= 0:
            used.add(int(i[r])); outs.append(int(sl[r])); ws.append(w10[r]); continue
        u = int(i[r])
        cand = [(Dm[u, e], e, s) for e, s in res if e not in used and Dm[u, e] < Nm[u]]
        STAT["n"] += 1
        if cand:
            _, e, s = min(cand); used.add(e); outs.append(s); ws.append(w10[r]); STAT["used"] += 1
    ws = np.array(ws, np.float32)
    if RENORM:
        ws = ws / ws.sum()
    pad = K - len(outs)
    outs += [outs[0]] * pad; ws = np.concatenate([ws, np.zeros(pad, np.float32)])
    base = type(sw).__mro__[1]
    y = (base.__call__(sw, x, mx.array(outs, mx.uint32).reshape(1, 1, K)) * mx.array(ws)[None, None, :, None].astype(x.dtype)).sum(axis=-2)
    return y + attention_gate(self.shared_expert(x), self.shared_expert_gate(x))


def step_forward_bild(rt, ids, pos, score, tau):
    """SG.step_forward with BiLD-style fallback: each 1-row frugal step's next-token entropy decides whether the token is
    rolled back (MTPLX rollback_after_verify, verified_tokens=1) and recomputed with the whole top-10 (MODE exact).
    Counts re-runs and SSD loads (ext misses) per decode step. The prefill's last token stays frugal."""
    from mtplx import generation as G
    from mtplx.attention_context import attention_phase
    from mtplx.cache_state import rollback_after_verify, snapshot_untrimmable_cache_lazy
    p0 = int(pos[0])
    ps = G.restore_or_prefill_prompt_state(rt, [int(x) for x in ids[:p0 + 1]], base_hidden_variant=None,
                                           mtp_history_policy="cycle", capture_hidden=False)
    cache, logits, out = ps.trunk_cache, ps.logits, []
    for t in pos:
        if t > p0:
            m0 = ext.stats()["misses"]
            snap = snapshot_untrimmable_cache_lazy(cache)
            x = mx.array([[int(ids[t])]])
            with attention_phase("ar_decode"):
                logits = rt.forward_ar(x, cache=cache)[:, -1, :]
            lp = logits.astype(mx.float32); lp = lp - mx.logsumexp(lp, axis=-1, keepdims=True)
            ent = float(-(mx.exp(lp) * lp).sum())
            if ent > tau:
                rollback_after_verify(cache, snap, verified_tokens=1)
                MODE["exact"] = True
                try:
                    with attention_phase("ar_decode"):
                        logits = rt.forward_ar(x, cache=cache)[:, -1, :]
                    mx.eval(logits)
                finally:
                    MODE["exact"] = False
                STAT["rerun"] = STAT.get("rerun", 0) + 1
            STAT["steps"] = STAT.get("steps", 0) + 1
            STAT["loads"] = STAT.get("loads", 0) + ext.stats()["misses"] - m0
        r = score(np.array([t]), logits)
        mx.eval(r)
        out.append({k: np.array(v) for k, v in r.items()})
    return {k: np.concatenate([o[k] for o in out]) for k in out[0]}
