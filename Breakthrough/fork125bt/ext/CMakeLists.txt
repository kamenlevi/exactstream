cmake_minimum_required(VERSION 3.27)
project(bigmoe_ext LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python 3.12 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -c "import mlx.core as m, os; print(os.path.join(os.path.dirname(m.__file__), 'share', 'cmake', 'MLX'))"
                OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE MLX_DIR)
find_package(MLX CONFIG REQUIRED)
if(NOT DEFINED NB_CMAKE_DIR)   # MLX 0.32.2 uses nanobind ABI v21 (vendored 2.15); 0.32.4 needs v22 (nanobind 3.1)
  set(NB_CMAKE_DIR "${CMAKE_SOURCE_DIR}/../.nb215/nanobind/cmake")
endif()
set(nanobind_DIR "${NB_CMAKE_DIR}")
find_package(nanobind CONFIG REQUIRED)
nanobind_add_module(bigmoe_ext NB_STATIC NOMINSIZE NB_DOMAIN mlx bigmoe_ext.cpp)
target_link_libraries(bigmoe_ext PRIVATE mlx)
